See It In Action

From Code to App in Minutes

my_app.bas 
' A simple temperature converter
let f# = form#("Converter", 320, 240)

label#(f#, "Celsius:", 20, 20)
let e# = edit#(f#, 20, 50, 280, 36)
let b# = button#(f#,"Convert",20,100,280,44)
let r# = label#(f#, "", 20, 160)

button_onclick#(b#, "doConvert")
form_show(f#)

function doConvert(sender#)
  local c, f
  let c = val(edit_text$(e#))
  let f = c * 9 / 5 + 32
  label_text#(r#, str$(c)+"°C = "+str$(f)+"°F")
endfunction
